// Import PKI.js library for handling X.509 certificates and structures
import * as pkijs from "pkijs";
// Import ASN.1.js library for encoding and decoding ASN.1 binary structures
import * as asn1js from "asn1js";
import { leafHASH, nodeHASH, stateHASH_raw } from "./hash_utils.js";

const DSA_ALGORITHM = "ML-DSA-65";
const KEM_ALGORITHM = "ML-KEM-768";
const ML_DSA_65_OID = "2.16.840.1.101.3.4.3.18";
const APP_NAME = "PQC File Sharing System";

/**
 * Converts a PEM-formatted string (such as certificates or keys) into a binary ArrayBuffer.
 * @param {string} pem - The PEM-formatted string containing headers, footers, and base64 data.
 * @returns {ArrayBuffer} The raw binary representation of the PEM content.
 */
function pemToBuffer(pem) {
    // Strip out PEM headers, footers, line breaks, and all whitespace characters
    const b64 = pem.replace(/-----(BEGIN|END)[^-]+-----|\s+/g, '');
    // Decode the base64 string into a binary string
    const binStr = atob(b64);
    const len = binStr.length;
    // Allocate a typed array for the binary bytes
    const bytes = new Uint8Array(len);
    // Populate the byte array from the binary string character codes
    for (let i = 0; i < len; i++) {
        bytes[i] = binStr.charCodeAt(i);
    }
    // Return the underlying ArrayBuffer
    return bytes.buffer;
}

/**
 * Encapsulates a raw binary buffer into a standard PEM-formatted text block.
 * @param {ArrayBuffer} buffer - The raw binary data to wrap.
 * @param {string} [label="PRIVATE KEY"] - The label to display in the PEM header and footer.
 * @returns {string} The formatted PEM text string.
 */
function BufferToPem(buffer, label = "PRIVATE KEY") {
    // Convert the ArrayBuffer bytes into a base64 encoded string
    const base64 = btoa(String.fromCharCode(...new Uint8Array(buffer)));
    // Format the base64 string by inserting a newline character every 64 characters
    const formatted = base64.match(/.{1,64}/g).join('\n');
    // Return the complete PEM string wrapped with header and footer tags
    return `-----BEGIN ${label}-----\n${formatted}\n-----END ${label}-----`;
}

/**
 * Parses an X.509 certificate object from a PEM-formatted string.
 * @param {string} pem - The PEM-formatted X.509 certificate string.
 * @returns {pkijs.Certificate} The parsed PKI.js Certificate object instance.
 */
export function decodeCertFromPem(pem) {
    // Convert the PEM string into a raw binary buffer
    const buf = pemToBuffer(pem);
    // Parse the ASN.1 schema structure and construct a PKI.js Certificate object
    const asn1 = asn1js.fromBER(buf);
    if (asn1.offset === (-1) || !asn1.result) {
        throw new Error("ASN.1 parsing failed. The binary data is not a valid DER-encoded certificate.");
    }

    // 5. Map to pkijs Certificate object safely
    return new pkijs.Certificate({ schema: asn1.result });
    // const cert = new pkijs.Certificate({ schema: asn1js.fromBER(buf).result });
    // // Return the initialized certificate object
    // return cert;
}

/**
 * Extracts and imports the Admin's ML-DSA public key from an X.509 certificate.
 * @param {pkijs.Certificate} adminCert - The Admin CA's certificate object.
 * @returns {Promise<CryptoKey>} The imported ML-DSA public CryptoKey configured for signature verification.
 */
export async function extractAdminDsaPK(adminCert) {
    // Export the SubjectPublicKeyInfo (spki) ASN.1 block from the certificate
    const adminpkcs8Buffer = adminCert.subjectPublicKeyInfo.toSchema().toBER(false);
    // Import the spki buffer into the ML-DSA provider engine as a verify-capable CryptoKey
    const adminDsaKey = await mldsa.importPK(
        "spki",
        adminpkcs8Buffer,
        { name: "ML-DSA-65" }
    );
    // Return the imported administrative public key
    return adminDsaKey;
}

/**
 * Extracts and imports a User's ML-KEM public key from their X.509 certificate.
 * @param {pkijs.Certificate} userCert - The user's certificate object.
 * @returns {Promise<CryptoKey>} The imported ML-KEM public CryptoKey configured for encapsulation.
 */
export async function extractUserKemPK(userCert) {
    // Extract the SubjectPublicKeyInfo (spki) sequence from the user certificate
    const userpkcs8Buffer = userCert.subjectPublicKeyInfo.toSchema().toBER(false);
    // Import the raw spki bytes into the ML-KEM provider engine for key encapsulation

    const bytes = new Uint8Array(userpkcs8Buffer);

    // Import the binary SPKI data into the ML-KEM engine for encapsulation operations
    const publicKey = await mlkem.importPK(
        "spki",
        bytes,
        { name: "ML-KEM-768" }
    );
    // Return the imported private key object
    return publicKey;
}

/**
 * Verifies the cryptographic signature of a user certificate using the Admin's ML-DSA public key.
 * @param {pkijs.Certificate} userCert - The user's certificate to verify.
 * @param {CryptoKey} adminDsaKey - The Admin's ML-DSA verification public key.
 * @returns {Promise<boolean>} True if the signature is valid, false otherwise.
 */
export async function verifyUserCert(userCert, adminDsaKey) {
    // 1. 获取最原始的 To-Be-Signed (TBS) 字节数组
    // 在 pkijs 中，解析证书后，原始的 TBS 字节（Uint8Array）直接保存在 userCert.tbs 属性中
    const userTbs = userCert.tbs;
    
    if (!userTbs) {
        console.error("🚨 无法从证书中提取 TBS 数据，证书结构可能已损坏。");
        return false;
    }

    // 2. 提取签名值 (Signature Value)
    const userSig = new Uint8Array(userCert.signatureValue.valueBlock.valueHexView);

    // 3. 调用 Web Crypto API 进行 ML-DSA 验签
    try {
        const isUserValid = await mldsa.verify(
            adminDsaKey, 
            userSig.buffer, 
            userTbs
        );

        // 4. 判断验签结果
        if (!isUserValid) {
            console.error("Failed to verify the certificate. It is not issued by Admin or illegally modified.");
            return false;
        }

        // console.log("User certificate verification passed.");
        return true;
        
    } catch (e) {
        console.error("Exception occurs when verification:", e.message);
        return false;
    }
}

/**
 * Encapsulates a shared secret using a recipient's ML-KEM public key.
 * @param {CryptoKey} kemPK - The recipient's ML-KEM public key.
 * @returns {Promise<{ct: ArrayBuffer, ss: ArrayBuffer}>} The generated ciphertext and shared secret.
 */
export async function mlkem_enc(kemPK) {
    // Invoke the ML-KEM encapsulation routine to produce ciphertext and a shared key
    const { ct, ss } = await mlkem.encapsulate({ name: "ML-KEM-768" }, kemPK);
    // Return the ciphertext and symmetric key mapping object
    return { ct, ss }
}

/**
 * Decapsulates a ciphertext using an ML-KEM private key to recover the shared secret.
 * @param {ArrayBuffer} sym_key_CT - The encapsulation ciphertext.
 * @param {CryptoKey} kemSK - The recipient's ML-KEM private key.
 * @returns {Promise<ArrayBuffer>} The recovered shared symmetric key buffer.
 */
export async function mlkem_dec(sym_key_CT, kemSK) {
    // Invoke the ML-KEM decapsulation routine using the private key and ciphertext
    const recovered_sym_key = await mlkem.decapsulate(
        { name: "ML-KEM-768" },
        kemSK,
        sym_key_CT
    );
    // Return the recovered shared secret buffer
    return recovered_sym_key;
}

/**
 * Creates an X.500 name while preserving one RDN set per supplied attribute.
 *
 * @param {Array<{type: string, value: string}>} attributes Name attributes.
 * @returns {pkijs.RelativeDistinguishedNames} Encoded distinguished name.
 */
function createDistinguishedName(attributes) {
    const schema = new asn1js.Sequence({
        value: attributes.map(({ type, value }) => new asn1js.Set({
            value: [new pkijs.AttributeTypeAndValue({
                type,
                value: new asn1js.Utf8String({ value })
            }).toSchema()]
        }))
    });
    return new pkijs.RelativeDistinguishedNames({ schema });
}

/**
 * Generates the Java-compatible positive 128-bit certificate serial number.
 * The low 32 bits contain the certificate index and the upper 96 bits are
 * cryptographically random.
 *
 * @param {number} certInd Non-negative signed Java integer certificate index.
 * @returns {asn1js.Integer} DER INTEGER serial number.
 */
function createCertificateSerial(certInd) {
    if (!Number.isInteger(certInd) || certInd < 0 || certInd > 0x7fffffff) {
        throw new RangeError("certInd must be a non-negative signed 32-bit integer");
    }

    const serial = globalThis.crypto.getRandomValues(new Uint8Array(16));
    new DataView(serial.buffer).setUint32(12, certInd, false);

    // ASN.1 INTEGER is signed. Prefix a zero byte when the random high bit is
    // set so the Java BigInteger serial remains positive.
    const positiveSerial = (serial[0] & 0x80) === 0
        ? serial
        : new Uint8Array([0, ...serial]);
    return new asn1js.Integer({ valueHex: positiveSerial.buffer });
}

/**
 * Issues an ML-DSA-65-signed X.509 user certificate containing an ML-KEM-768
 * SubjectPublicKeyInfo, following the Java `issueCertForUser` layout.
 *
 * @param {string} userId User identifier placed in the subject common name.
 * @param {number} certInd Certificate index stored in the serial number's low 32 bits.
 * @param {CryptoKey|Object|string} userKemPK ML-KEM public key object or SPKI PEM.
 * @param {CryptoKey|Object|string} adminDsaSK Admin ML-DSA private key object or PKCS#8 PEM.
 * @param {number|Date} [createTime=Date.now()] Validity start time.
 * @param {number} [validTime=86400000] Validity duration in milliseconds.
 * @returns {Promise<pkijs.Certificate>} Signed certificate compatible with
 * {@link extractUserKemPK}, {@link verifyUserCert}, and {@link exportCertToPem}.
 */
export async function issueUserCert(
    userId,
    certInd,
    userKemPK,
    adminDsaSK,
    createTime = Date.now(),
    validTime = 24 * 60 * 60 * 1000
) {
    if (typeof userId !== "string" || userId.length === 0) {
        throw new TypeError("userId must be a non-empty string");
    }
    if (!Number.isFinite(validTime) || validTime <= 0) {
        throw new RangeError("validTime must be a positive number of milliseconds");
    }

    const notBefore = createTime instanceof Date
        ? new Date(createTime.getTime())
        : new Date(createTime);
    const notAfter = new Date(notBefore.getTime() + validTime);
    if (Number.isNaN(notBefore.getTime()) || Number.isNaN(notAfter.getTime())) {
        throw new RangeError("Certificate validity dates are invalid");
    }

    const kemPublicKey = typeof userKemPK === "string"
        ? importKemPkFromPem(userKemPK)
        : userKemPK;
    const signingKey = typeof adminDsaSK === "string"
        ? importDsaSKFromPem(adminDsaSK)
        : adminDsaSK;
    if (!kemPublicKey || kemPublicKey.type !== "public") {
        throw new TypeError("userKemPK must be an ML-KEM public key or SPKI PEM");
    }
    if (kemPublicKey.algorithm?.name !== KEM_ALGORITHM) {
        throw new TypeError(`userKemPK must use ${KEM_ALGORITHM}`);
    }
    if (!signingKey || signingKey.type !== "private") {
        throw new TypeError("adminDsaSK must be an ML-DSA private key or PKCS#8 PEM");
    }
    if (signingKey.algorithm?.name !== DSA_ALGORITHM) {
        throw new TypeError(`adminDsaSK must use ${DSA_ALGORITHM}`);
    }

    const spkiBuffer = await mlkem.exportPK("spki", kemPublicKey);
    const spkiAsn1 = asn1js.fromBER(spkiBuffer);
    if (spkiAsn1.offset === -1 || !spkiAsn1.result) {
        throw new Error("Unable to encode the user's ML-KEM SubjectPublicKeyInfo");
    }

    const signatureAlgorithm = new pkijs.AlgorithmIdentifier({
        algorithmId: ML_DSA_65_OID
    });
    const certificate = new pkijs.Certificate({
        version: 2,
        serialNumber: createCertificateSerial(certInd),
        signature: signatureAlgorithm,
        issuer: createDistinguishedName([
            { type: "2.5.4.3", value: "Admin MLDSA CA" },
            { type: "2.5.4.10", value: APP_NAME }
        ]),
        notBefore: new pkijs.Time({ value: notBefore }),
        notAfter: new pkijs.Time({ value: notAfter }),
        subject: createDistinguishedName([
            { type: "2.5.4.3", value: userId }
        ]),
        subjectPublicKeyInfo: new pkijs.PublicKeyInfo({ schema: spkiAsn1.result }),
        signatureAlgorithm
    });

    const tbs = certificate.encodeTBS().toBER(false);
    const signature = await mldsa.sign(signingKey, tbs);
    certificate.tbsView = new Uint8Array(tbs);
    certificate.signatureValue = new asn1js.BitString({ valueHex: signature });

    // Parse the completed DER again so `tbsView` is exactly the byte sequence
    // consumed by verifyUserCert and Java X.509 implementations.
    const certificateDer = certificate.toSchema(false).toBER(false);
    const certificateAsn1 = asn1js.fromBER(certificateDer);
    if (certificateAsn1.offset === -1 || !certificateAsn1.result) {
        throw new Error("Unable to encode the issued user certificate");
    }
    return new pkijs.Certificate({ schema: certificateAsn1.result });
}

/**
 * Exports a PKI.js certificate as a PEM block.
 *
 * @param {pkijs.Certificate} cert Certificate to export.
 * @param {string} [label="USER CERTIFICATE"] PEM label.
 * @returns {string} DER certificate wrapped as PEM.
 */
export function exportCertToPem(cert, label = "USER CERTIFICATE") {
    if (!(cert instanceof pkijs.Certificate)) {
        throw new TypeError("cert must be a PKI.js Certificate");
    }
    return BufferToPem(cert.toSchema(false).toBER(false), label);
}

/**
 * Checks whether an X.509 certificate has expired based on the current system time.
 * @param {pkijs.Certificate} cert - The certificate object to evaluate.
 * @returns {boolean} True if the certificate is expired, false if it is currently valid.
 */
export function isCertExpired(cert) {
    // Capture the current local date and time instance
    const now = new Date();
    // Validate if the current timestamp falls strictly within the validity boundary window
    if (now >= cert.notBefore.value && now <= cert.notAfter.value) {
        return false; // Return false because the certificate has not expired yet
    } else {
        return true;  // Return true because the certificate is expired or not yet active
    }
}

/**
 * Validates whether a given certificate contains the CA (Certification Authority) basic constraints flag.
 * @param {pkijs.Certificate} rootCert - The certificate object to check.
 * @returns {boolean} True if the certificate is a valid CA, false otherwise.
 */
export function isCACert(rootCert) {
    let isCA = false;

    // Check if the certificate contains any extensions
    if (rootCert.extensions && Array.isArray(rootCert.extensions)) {
        for (const ext of rootCert.extensions) {
            // Check if the extension OID matches the Basic Constraints extension identifier (2.5.29.19)
            if (ext.extnID === "2.5.29.19") {
                // Note: ext.parsedValue is already parsed by pkijs, so we can directly access its properties.
                // If ext.parsedValue is a BasicConstraints object, it directly exposes the 'cA' property.
                if (ext.parsedValue && ext.parsedValue.cA === true) {
                    isCA = true;
                    break; // Exit loop early once CA confirmation is found
                }
            }
        }
    }

    // Log a warning message if the CA flag is missing or disabled
    if (!isCA) {
        console.warn("Critical Error: this cert is issued by an illegal CA or lacks the CA basic constraint.");
    }

    // Return the evaluation result flag
    return isCA;
}

/**
 * Verifies a serialized Merkle-tree membership proof and its signed state hash.
 *
 * @param {string} proofJsonStr JSON-encoded proof containing the certificate,
 * proof nodes, previous state hash, timestamp, duration, and state signature.
 * @param {CryptoKey|Object} adminDsaPK Admin ML-DSA public key used for verification.
 * @param {number} time Timestamp at which membership should be checked.
 * @returns {Promise<boolean>} Whether the proof is current and its signature is valid.
 * @throws {SyntaxError} If `proofJsonStr` is not valid JSON.
 */

export async function verifyMemberProof(proofJsonStr, adminDsaPK, time){
    if (proofJsonStr == "")
        return false;
    const proof = JSON.parse(proofJsonStr);
    if(time >= proof.timestamp + proof.duration || time < proof.timestamp){
        const t = new Date(time);
        console.log("This membership proof is expired or not recorded at time " + t);
        return false;
    }

    // get hash of cert
    const certHash = leafHASH(pemToBuffer(proof.userCertPem));
    let preHash = certHash;
    // compute the root
    proof.proofNodes.forEach(node => {
        if(node.isLeft){
            preHash = nodeHASH(node.hash, preHash);
        }else{
            preHash = nodeHASH(preHash, node.hash);
        }
    });
    const rootHash = preHash;
    // compute state hash
    const stateHash = stateHASH_raw(rootHash, proof.preStateHash, proof.timestamp);
    const sigBuffer = pemToBuffer(proof.curStateSig);
    const isValid = await mldsa.verify(
        adminDsaPK,
        sigBuffer,
        stateHash
    );
    return isValid;
}

/**
 * Encapsulates an AES key for an ML-KEM recipient.
 *
 * @param {CryptoKey} kemPK - The recipient's ML-KEM public key.
 * @param {ArrayBuffer} aesKey - The symmetric AES key to be encrypted.
 * @returns {Promise<{ssCT: ArrayBuffer, aesCT: ArrayBuffer}>} The generated shared secret ciphertext and AES key ciphertext.
 */
export async function mlkem_enc_with_aesKey(kemPK, aesKey) {
    // Invoke the ML-KEM encapsulation routine to produce ciphertext and a shared key
    const { ct:ssCT, ss:ss } = await pqc_funcs.mlkem_enc(kemPK);
    // Encrypt the provided AES key using the derived shared session key
    const aesCT = await aes_enc(aesKey, ss);
    // Return the ciphertext and symmetric key mapping object
    return { ssCT, aesCT }
}

/**
 * Recovers an AES key from its ML-KEM and AES-GCM ciphertexts.
 *
 * @param {CryptoKey} kemSK - The recipient's ML-KEM private key.
 * @param {ArrayBuffer} ssCT - The ML-KEM ciphertext.
 * @param {ArrayBuffer} aesCT - The AES ciphertext.
 * @returns {Promise<ArrayBuffer>} The decrypted AES key bytes.
 */
export async function mlkem_dec_with_aesKey(kemSK, ssCT, aesCT) {
    // Invoke the ML-KEM decapsulation routine to recover the shared session key
    const recoveredSS = await pqc_funcs.mlkem_dec(ssCT, kemSK);
    // Decrypt the AES key ciphertext using the recovered shared session key
    const recoveredAESKey = await aes_dec(aesCT, recoveredSS);
    // Return the decrypted AES key bytes
    return recoveredAESKey;
}

/**
 * Re-encrypts an encapsulated AES key for a new ML-KEM recipient.
 *
 * @param {CryptoKey} kemSK - The original ML-KEM private key used to decapsulate the original ciphertext.
 * @param {ArrayBuffer} ssCT - The ML-KEM ciphertext, which encapsulates the shared session key ss.
 * @param {ArrayBuffer} aesCT - The AES key's ciphertext, encrypted by ss using AES-GCM.
 * @param {CryptoKey} newKemPK - The new ML-KEM public key to re-encapsulate the recovered AES key.
 * @returns {Promise<{ssCT: ArrayBuffer, aesCT: ArrayBuffer}>}
 */
export async function mlkem_re_enc_with_aesKey(kemSK, ssCT, aesCT, newKemPK) {
    // Decapsulate the original ML-KEM ciphertext to recover the encrypted AES key.
    const recoveredAESKey = await mlkem_dec_with_aesKey(kemSK, ssCT, aesCT);
    // Re-encapsulate the AES key with the new ML-KEM public key
    return await mlkem_enc_with_aesKey(newKemPK, recoveredAESKey);
}

/**
 * Generates a cryptographically secure 256-bit AES key.
 *
 * @returns {Promise<Uint8Array>} A random 256-bit AES key in Uint8Array format for symmetric encryption.
 */
export async function aes_random_key_gen() {
    // Generate a new random 256-bit AES key for symmetric encryption
    const aesKey = globalThis.crypto.getRandomValues(new Uint8Array(32));
    return aesKey;
}

/**
 * Encrypts a message buffer using the AES-GCM symmetric encryption scheme.
 * @param {Uint8Array} msg - The plaintext message bytes to encrypt.
 * @param {Uint8Array} key - The AES-GCM symmetric key.
 * @returns {Promise<Uint8Array>} A combined buffer containing the random IV and the resulting ciphertext.
 */
export async function aes_enc(msg, key) {
    // 1. Import the raw symmetric key into a Web Crypto API CryptoKey object.
    // Note: The AES key length must be exactly 16 bytes (128-bit), 24 bytes (192-bit), or 32 bytes (256-bit).
    const cryptoKey = await globalThis.crypto.subtle.importKey(
        "raw",                     // Format: Use "raw" for standard binary cryptographic keys.
        key,                       // The raw key material (must be an ArrayBuffer or Uint8Array).
        { name: "AES-GCM" },       // Target cryptographic algorithm.
        false,                     // Extractability: Set to false to prevent the key material from being exported.
        ["encrypt", "decrypt"]     // Key usages: Permitted operations for this key (encrypt and decrypt).
    );

    // Generate a secure cryptographically random 12-byte (96-bit) Initialization Vector (IV)
    const iv = globalThis.crypto.getRandomValues(new Uint8Array(12));
    // Execute native browser hardware-accelerated AES-GCM encryption
    const ciphertext = await globalThis.crypto.subtle.encrypt(
        { name: "AES-GCM", iv: iv },
        cryptoKey,
        msg
    );

    // Allocate a unified buffer block to house both the IV prefix and the encrypted payload data
    const combined = new Uint8Array(iv.byteLength + ciphertext.byteLength);
    // Copy the IV bytes into the beginning of the combined buffer
    combined.set(iv, 0);
    // Copy the ciphertext bytes immediately following the IV section
    combined.set(new Uint8Array(ciphertext), iv.byteLength);

    // Return the complete combined binary payload block
    return combined;
}

/**
 * Decrypts a combined IV-and-ciphertext buffer using AES-GCM.
 * @param {Uint8Array} combinedData - The combined payload containing both the IV prefix and ciphertext.
 * @param {Uint8Array} key - The AES-GCM symmetric key.
 * @returns {Promise<Uint8Array>} The recovered decrypted plaintext byte array.
 */
export async function aes_dec(combinedData, key) {
    // 1. Import the raw symmetric key into a Web Crypto API CryptoKey object.
    // Note: The AES key length must be exactly 16 bytes (128-bit), 24 bytes (192-bit), or 32 bytes (256-bit).
    const cryptoKey = await globalThis.crypto.subtle.importKey(
        "raw",                     // Format: Use "raw" for standard binary cryptographic keys.
        key,                       // The raw key material (must be an ArrayBuffer or Uint8Array).
        { name: "AES-GCM" },       // Target cryptographic algorithm.
        false,                     // Extractability: Set to false to prevent the key material from being exported.
        ["encrypt", "decrypt"]     // Key usages: Permitted operations for this key (encrypt and decrypt).
    );

    // Isolate the leading 12 bytes to serve as the Initialization Vector (IV)
    const iv = combinedData.subarray(0, 12);
    // Isolate all remaining trailing bytes to serve as the encrypted payload ciphertext
    const ciphertext = combinedData.subarray(12);

    // Execute native browser AES-GCM decryption using the extracted IV and key
    const decrypted = await globalThis.crypto.subtle.decrypt(
        { name: "AES-GCM", iv: iv },
        cryptoKey,
        ciphertext
    );

    // Return the decrypted output wrapped in a Uint8Array view
    return new Uint8Array(decrypted);
}
