 * Executes a simulated end-to-end cryptographic workflow integrating KEM, digital certificates, and AES encryption.
 * @tutorial quick-start
 * @returns {Promise<void>}
 */
export async function test_work_flow() {
    // Generate a fresh client-side ML-KEM key pair for original enc-dec; another pair of kem keys (extractedKemPK, extractedKemSK) will be extracted for re-encryption
    const { publicKey: kemPK, privateKey: kemSK } = await pqc_funcs.kemKeyGen();

    // +++++++++Send CSR to Admin+++++++++
    // Export the KEM public key into PEM format for CSR inclusion
    const kemPKPem = await pqc_funcs.exportKemPKToPem(kemPK);
    // Construct a simulated certificate signing request (CSR) payload block
    const csr = { "user_id": 123, "kemPK": kemPKPem };

    // +++++++++Get User Cert From Server+++++++++
    // Simulate obtaining a signed user certificate string from the administrative authority
    const userCert = pqc_funcs.decodeCertFromPem(userCertPem);

    // +++++++++Check Expiration of User Cert+++++++++
    // Evaluate if the user certificate is active or has expired
    if (!pqc_funcs.isCertExpired(userCert))
        console.log("User cert is not expired");
    else
        console.log("User cert is expired");

    // +++++++++Get Admin Cert From Server+++++++++
    // Simulate obtaining the self-issued Admin root CA certificate
    const adminCert = pqc_funcs.decodeCertFromPem(adminCertPem);

    // +++++++++Check Correctness of Admin Cert+++++++++
    // 1. Evaluate if the admin root certificate is active or expired
    if (!pqc_funcs.isCertExpired(adminCert))
        console.log("Admin cert is not expired");
    else
        console.log("Admin cert is expired");

    // 2. Validate that the admin certificate possesses valid CA permissions
    pqc_funcs.isCACert(adminCert);

    // +++++++++Get Admin MLDSA PK+++++++++
    // Extract the administrative ML-DSA public key from its certificate for proof verification and user certificate validation
    const adminDsaPK = await pqc_funcs.extractAdminDsaPK(adminCert);
    console.log("Loaded Admin dsa PK", adminDsaPK);

    // +++++++++Get User MLKEM PK+++++++++
    // Extract the embedded ML-KEM public key from the user certificate for continuity encryption/re-encryption operations
    const extractKemPK = await pqc_funcs.extractUserKemPK(userCert);
    console.log("Loaded User kem PK", extractKemPK);

    // +++++++++Verify User Cert+++++++++
    // Perform signature verification on the user certificate using the admin public key
    if (await pqc_funcs.verifyUserCert(userCert, adminDsaPK)) {
        console.log("User certificate verification passed");
    } else {
        console.log("User certificate verification failed");
    }

    // +++++++++Verify Membership Proof+++++++++
    // The timestamp is specified by the user to verify the certificate's membership validity at a specific point in time.
    const time = 160; // long milliseconds
    if(await pqc_funcs.verifyMemberProof(proofJSONStr, adminDsaPK, time))
        console.log("Proof Passed!!!!!!!");
    else
        console.log("Proof Unpassed!!!!!!!");

    // +++++++++Test PQC encryption decryption+++++++++
    // +++++++++ENC MSG+++++++++
    // Prepare a plain text message payload to encrypt
    const msg = new TextEncoder().encode("Hello world");
    // Sample a 256-bit AES symmetric key for encrypting the message payload
    const aesKey = await pqc_funcs.aes_random_key_gen();
    // Encapsulate the aesKey using the original KEM public key kemPK
    const { ssCT, aesCT } = await pqc_funcs.mlkem_enc_with_aesKey(kemPK, aesKey);
    // Encrypt the target message payload using the aesKey with AES-GCM symmetric encryption
    const msgCT = await pqc_funcs.aes_enc(msg, aesKey);

    // +++++++++Re-ENC MSG+++++++++
    // Re-encapsulate the aesKey using the new KEM public key extractKemPK
    const { ssCT: reEncSSCT, aesCT: reEncAESCT } = await pqc_funcs.mlkem_re_enc_with_aesKey(kemSK, ssCT, aesCT, extractKemPK);

    // +++++++++DEC MSG+++++++++
    // Decrypt using the original KEM private key (kemSK)
    const recoveredAesKey = await pqc_funcs.mlkem_dec_with_aesKey(kemSK, ssCT, aesCT);
    for(var i = 0; i < 32; i++){
        if (aesKey.at(i) != recoveredAesKey.at(i)){
            console.error("Original KEM Decrypt Failed", i);
            break;
        }
    }

    // +++++++++DEC MSG for RE-ENC+++++++++
    // Load the KEM SK for the re-encrypted ciphertext's decryption 
    const extractedKemSK = pqc_funcs.importKemSkFromPem(userSKPem);
    console.log("Loaded User kem SK", extractedKemSK);
    // Decrypt using the extracted KEM SK for the re-encrypted ciphertext
    const recoveredAesKey2 = await pqc_funcs.mlkem_dec_with_aesKey(extractedKemSK, reEncSSCT, reEncAESCT);
    for(var i = 0; i < 32; i++){
        if (aesKey.at(i) != recoveredAesKey2.at(i)){
            console.error("Re-encrypted KEM Decrypt Failed", i);
            break;
        }
    }

    // Decrypt the transmission ciphertext bytes back into raw plaintext bytes using the recovered session key
    const decryptedMsgBytes = await pqc_funcs.aes_dec(msgCT, recoveredAesKey);

    // Decode the decrypted bytes and the original payload bytes into strings for final verification
    const decryptedText = new TextDecoder().decode(decryptedMsgBytes);
    const originalText = new TextDecoder().decode(msg);

    // Compare plaintexts to confirm successful end-to-end data transmission
    if (decryptedText === originalText) {
        console.log("Decrypt successfully:", decryptedText);
    } else {
        console.log("Decrypt failed");
    }
}
